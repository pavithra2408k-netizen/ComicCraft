async function createComic() { const idea = document.getElementById("storyIdea").value.trim(); if (idea === "") { alert("Please enter a story idea!"); return; } document.getElementById("comicArea").style.display = "block"; // Show loading text while waiting for Gemini response document.getElementById("panel1").innerText = "Generating story..."; document.getElementById("panel2").innerText = "Please wait..."; document.getElementById("panel3").innerText = "..."; document.getElementById("panel4").innerText = "..."; const apiKey = "AIza...."; try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: `Create a 4-panel comic story script based on this idea: "${idea}". Make it brief for each panel.` }] }] }) }); const data = await response.json(); const resultText = data.candidates[0].content.parts[0].text; // Display the generated text in Panel 1 document.getElementById("panel1").innerText = resultText; document.getElementById("panel2").innerText = "See Panel 1 for full script output."; document.getElementById("panel3").innerText = ""; document.getElementById("panel4").innerText = ""; } catch (error) { console.error(error); alert("Error generating comic. Please check your API key and network connection."); } }
